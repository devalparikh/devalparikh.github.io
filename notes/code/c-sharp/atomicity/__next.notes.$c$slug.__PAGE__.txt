<h1 id="domain-events">Domain Events</h1>
<p>Create (aka Raise) and Consume (aka Publish) events
<a href="https://www.youtube.com/watch?v=AHzWJ_SMqLo" target="_blank" rel="noopener noreferrer">https://www.youtube.com/watch?v=AHzWJ_SMqLo</a></p>
<p>Usage flow</p>
<ul>
<li>EntityFramework (EF) will track changes to the entity objects</li>
<li>Each entity object will have a list of domain events for the given entity</li>
<li>When performing ******<a href="/notes/code/c-sharp/atomicity/#outbox-pattern">Outbox Pattern</a> the entities changes tracked are fetched (via ChangeTracker) and obtain all domain events</li>
<li>Create outbox message for each domain event</li>
<li>Save all outbox message</li>
<li>EF generates sql command for all changes (original and new outbox messages changes) to ensure atomicity</li>
<li>Commits transaction</li>
</ul>
<h1 id="outbox-pattern">Outbox Pattern</h1>
<p>Intercept db action and modify db context to include another change in the same transaction
Then consume those intercepted atomic entries separately
<a href="https://www.youtube.com/watch?v=XALvnX7MPeo" target="_blank" rel="noopener noreferrer">https://www.youtube.com/watch?v=XALvnX7MPeo</a>
<a href="https://chatgpt.com/share/67da2996-db10-8001-a544-b71f1a56e105" target="_blank" rel="noopener noreferrer">https://chatgpt.com/share/67da2996-db10-8001-a544-b71f1a56e105</a></p>

    await _dbContext.SaveChangesAsync(); 
}


// 1) domain events


public class Order : AggregateRoot
{
    public string OrderNumber { get; private set; }

    public void PlaceOrder()
    {
        // Raise a domain event when an order is placed
        RaiseDomainEvent(new OrderPlacedEvent(this.Id));
    }
}

public class AggregateRoot : Entity
{
	// multiple domain events per object
	private List&#x3C;DomainEvents> _domainEvents == new();

	// add
	public RaiseDomainEvent(DomainEvent domainEvent);

	// get
	public GetDomainEvents();
	
	// clear
	public ClearDomainEvents();
}


// 2) interceptor


public class DomainEventsInterceptor : SaveChangesInterceptor
{
    public override async ValueTask&#x3C;int> SavedChangesAsync(
        SaveChangesCompletedEventData eventData, 
        int result, 
        CancellationToken cancellationToken = default)
    {
        var context = eventData.Context;
        if (context == null) return result;

        // Step 1: Get all domain events from AggregateRoots
        var domainEvents = context.ChangeTracker.Entries&#x3C;AggregateRoot>()
            .SelectMany(e => e.Entity.GetDomainEvents())
            .ToList();

        if (domainEvents.Any())
        {
            // Step 2: Convert domain events into Outbox Messages
            var outboxMessages = domainEvents.Select(e => new OutboxMessage(e)).ToList();

            // Step 3: Persist outbox messages
            await context.Set&#x3C;OutboxMessage>().AddRangeAsync(outboxMessages, cancellationToken);

            // Step 4: Call base method to persist outbox messages (and the original db changes)
            await base.SavedChangesAsync(eventData, result, cancellationToken);

            // Step 5: Clear domain events from AggregateRoot
            foreach (var entity in context.ChangeTracker.Entries&#x3C;AggregateRoot>())
            {
                entity.Entity.ClearDomainEvents();
            }
        }

        return result;
    }
}

<p>then a background job can consume the outbox messages</p>
<p>benefits:</p>
<ul>
<li>domain events decoupled events</li>
<li>outbox takes domain events and makes it atomic with db commit</li>
</ul>
<p>drawbacks:</p>
<ul>
<li>if process goes down while processing events -> need Stateful</li>
